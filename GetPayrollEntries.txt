		CONVERT(Varchar(20), Getdate(), 101), CONVERT(Varchar(20), GETDATE(), 101)
	FROM PayrollReport a
	WHERE  a.ClientCode = @ClientCode AND a.StartPayDate = @StartPayDate AND a.Endpaydate = @EndPayDate

		EmployeeName, EmpNo, HRDepartment, ClientCutoffNo, CONVERT(Varchar(20), GETDATE(), 101),
		DaysPresent, NightDiffOT, CASE WHEN DaysPresent = NULL THEN 0.00 ELSE (DaysPresent/8) END,

		HoursTardiness, TardinessDeduction, DaysAbsent, AbsentDeduction, TaxRefundPay, SickLeavePay, VacationLeavePay,

		GrossPay, CASE WHEN Ecolapay = NULL THEN 0.00 ELSE EcolaPay END, 
		CASE WHEN Uniform = NULL THEN 0.00 ELSE Uniform END, CASE WHEN ValewithClient = NULL THEN 0.00 ELSE ValewithClient END, 
		CASE WHEN CashAdvance = NULL THEN 0.00 ELSE CashAdvance END, CASE WHEN SSSEmployee = NULL THEN 0.00 ELSE SSSEmployee END,
